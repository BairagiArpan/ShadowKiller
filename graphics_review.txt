========== src/config/perf.ts ==========
/**
 * Watches frame times and says when to drop one graphics level. Pure and testable.
 * It judges windows of frames, ignores hitches (app switches, pauses) and never raises quality.
 */
export class PerfGovernor {
  private acc = 0;
  private frames = 0;
  private cooldown = 0;

  constructor(
    private readonly targetMs: number,
    private readonly windowMs = 2000,
    private readonly slack = 1.35,
  ) {}

  /** Feed one frame time in milliseconds. Returns true when quality should drop one level. */
  sample(ms: number): boolean {
    if (!(ms > 0) || ms > 100) return false;
    this.acc += ms;
    this.frames += 1;
    if (this.acc < this.windowMs) return false;
    const avg = this.acc / this.frames;
    this.acc = 0;
    this.frames = 0;
    if (this.cooldown > 0) { this.cooldown -= 1; return false; } // let a change settle for one window
    if (avg > this.targetMs * this.slack) { this.cooldown = 1; return true; }
    return false;
  }
}

import { PLAYER_ATTACKS, attackHitbox } from './combat';
import { defOf, enemyAttackDef } from './enemies';
import {
  FIGHTER_H, FX_TIME, MAX_ENEMIES, MAX_ENERGY, MAX_HAZARDS, MAX_PROJECTILES, SPECIAL,
} from './constants';
import { HAZARD_BASE, hazardPhase } from './hazards';
import { poseFor } from './pose';
import type { Arena, Fighter } from './types';
import type { AttackDef } from './combat';

/** Plain data the renderer needs each frame. No Skia/React imports here. */
export interface FighterFrame {
  x: number; y: number; facing: number; lean: number; weapon: number; squash: number;
  opacity: number; warn: number; visible: number; color: string;
  slashX: number; slashY: number; slashW: number; slashOn: number;
}
export interface ProjectileFrame { x: number; y: number; on: number }
export interface HazardFrame { x: number; w: number; opacity: number; color: string; on: number }
export interface RenderFrame {
  player: FighterFrame;
  enemies: FighterFrame[];         // always MAX_ENEMIES long
  projectiles: ProjectileFrame[];  // always MAX_PROJECTILES long
  hazards: HazardFrame[];          // always MAX_HAZARDS long
  fx: { x: number; y: number; k: number; on: number };
  playerHp: number; energy: number; enemyHp: number; // 0..1
  bossHp: number;                  // 0..1, or -1 when no boss is alive
}

const HIDDEN: FighterFrame = {
  x: 0, y: 0, facing: 1, lean: 0, weapon: 0, squash: 1, opacity: 0, warn: 0, visible: 0,
  color: '#000000', slashX: 0, slashY: 0, slashW: 0, slashOn: 0,
};
const HIDDEN_HAZARD: HazardFrame = { x: 0, w: 0, opacity: 0, color: '#000000', on: 0 };

function frameOf(
  f: Fighter, def: AttackDef | null, color: string, warn: boolean,
  slash: { x: number; w: number; on: number } | null,
): FighterFrame {
  const pose = poseFor(f, def);
  return {
    x: f.x, y: f.y, facing: f.facing, lean: pose.lean, weapon: pose.weapon, squash: pose.squash,
    opacity: f.state === 'dead' ? 0.35 : f.invuln > 0 ? 0.5 : 1,
    warn: warn ? 1 : 0, visible: 1,
    color: f.state === 'hurt' ? '#ffffff' : color,
    slashX: slash?.x ?? 0, slashY: f.y - FIGHTER_H * 0.8, slashW: slash?.w ?? 0, slashOn: slash?.on ?? 0,
  };
}

export function buildFrame(a: Arena, playerColor = '#7a2cff'): RenderFrame {
  const p = a.player;
  const pdef = p.state === 'attack' && p.attackKind ? PLAYER_ATTACKS[p.attackKind] : null;
  let pslash: { x: number; w: number; on: number } | null = null;
  if (pdef) {
    const b = attackHitbox(p, { ...pdef, reach: pdef.reach * a.perks.reach });
    if (b) pslash = { x: b.x, w: b.w, on: 0.8 };
  } else if (p.state === 'special' && p.t < 0.3) {
    pslash = { x: p.x - SPECIAL.radius, w: SPECIAL.radius * 2, on: 0.35 };
  }

  const enemies: FighterFrame[] = [];
  let hp = 0, maxHp = 0, bossHp = -1;
  for (let i = 0; i < MAX_ENEMIES; i++) {
    const e = a.enemies[i];
    if (!e) { enemies.push(HIDDEN); continue; }
    const d = defOf(e);
    const def = enemyAttackDef(d);
    const box = d.style === 'shot' ? null : attackHitbox(e, def);
    enemies.push(frameOf(e, def, d.color, e.state === 'telegraph',
      box ? { x: box.x, w: box.w, on: 0.8 } : null));
    hp += e.hp; maxHp += e.maxHp;
    if (e.boss && e.state !== 'dead') bossHp = e.hp / e.maxHp;
  }

  const projectiles = [];
  for (let i = 0; i < MAX_PROJECTILES; i++) {
    const pr = a.projectiles[i];
    projectiles.push(pr ? { x: pr.x, y: pr.y, on: 1 } : { x: 0, y: 0, on: 0 });
  }

  const hazards: HazardFrame[] = [];
  for (let i = 0; i < MAX_HAZARDS; i++) {
    const h = a.hazards[i];
    if (!h) { hazards.push(HIDDEN_HAZARD); continue; }
    const phase = hazardPhase(h, a.time);
    let opacity = 0.18;                                       // dormant: faint marking
    if (h.kind === 'ice') opacity = 0.5;
    else if (phase === 2) opacity = 0.95;                     // active: bright and solid
    else if (phase === 1) opacity = Math.floor(a.time * 8) % 2 === 0 ? 0.7 : 0.25; // warning: blinks
    hazards.push({ x: h.x, w: h.w, opacity, color: HAZARD_BASE[h.kind].color, on: 1 });
  }

  return {
    player: frameOf(p, pdef, playerColor, false, pslash),
    enemies, projectiles, hazards,
    fx: a.fx.on
      ? { x: a.fx.x, y: a.fx.y, k: Math.min(1, a.fx.age / FX_TIME), on: 1 }
      : { x: 0, y: 0, k: 1, on: 0 },
    playerHp: p.hp / p.maxHp,
    energy: a.energy / MAX_ENERGY,
    enemyHp: maxHp > 0 ? hp / maxHp : 0,
    bossHp,
  };
}


========== src/engine/particles.ts ==========
import { GROUND_Y, WORLD_W } from './constants';

export type AmbientStyle = 'motes' | 'ash' | 'snow' | 'embers' | 'void';
export interface Point { x: number; y: number }

export interface Pool {
  cap: number;
  x: Float32Array; y: Float32Array; vx: Float32Array; vy: Float32Array; life: Float32Array;
}

/** Visual-only. Owns its own seeded random generator so it can never affect combat. */
export interface ParticleSystem {
  style: AmbientStyle;
  ambient: Pool; impact: Pool; hurt: Pool;
  ambientActive: number; // how many ambient particles are currently drawn (<= ambient.cap)
  perHit: number;        // impact particles per hit
  heads: [number, number];
  seed: number;
}

export const NO_POINTS: Point[] = [{ x: -50, y: -50 }]; // drawn off-screen when a group is empty

const RISING: Record<AmbientStyle, boolean> = { motes: true, ash: false, snow: false, embers: true, void: true };

function makePool(cap: number): Pool {
  const n = Math.max(0, Math.floor(cap));
  return {
    cap: n, x: new Float32Array(n), y: new Float32Array(n),
    vx: new Float32Array(n), vy: new Float32Array(n), life: new Float32Array(n),
  };
}

function rand(ps: ParticleSystem): number {
  ps.seed = (Math.imul(ps.seed, 1664525) + 1013904223) >>> 0;
  return ps.seed / 4294967296;
}

function spawnAmbient(ps: ParticleSystem, i: number, anywhere: boolean): void {
  const p = ps.ambient;
  p.x[i] = rand(ps) * WORLD_W;
  p.y[i] = anywhere ? rand(ps) * GROUND_Y : RISING[ps.style] ? GROUND_Y : -4;
  const r = rand(ps);
  switch (ps.style) {
    case 'motes':  p.vx[i] = (r - 0.5) * 16;      p.vy[i] = -(6 + rand(ps) * 8); break;
    case 'ash':    p.vx[i] = 12 + r * 14;         p.vy[i] = 14 + rand(ps) * 14; break;
    case 'snow':   p.vx[i] = (r - 0.5) * 20;      p.vy[i] = 25 + rand(ps) * 25; break;
    case 'embers': p.vx[i] = (r - 0.5) * 28;      p.vy[i] = -(30 + rand(ps) * 40); break;
    case 'void':   p.vx[i] = (r - 0.5) * 20;      p.vy[i] = -(4 + rand(ps) * 10); break;
  }
  p.life[i] = 1;
}

export function createParticles(
  style: AmbientStyle, ambientCount: number, impactCap: number, perHit: number, seed = 12345,
): ParticleSystem {
  const ps: ParticleSystem = {
    style, ambient: makePool(ambientCount), impact: makePool(impactCap),
    hurt: makePool(Math.floor(impactCap / 2)),
    ambientActive: Math.max(0, Math.floor(ambientCount)), perHit: Math.max(0, Math.floor(perHit)),
    heads: [0, 0], seed: seed >>> 0,
  };
  for (let i = 0; i < ps.ambient.cap; i++) spawnAmbient(ps, i, true);
  return ps;
}

/** Lower (never raise past the pool size) the drawn amount at runtime, e.g. after an auto-downgrade. */
export function setParticleLimits(ps: ParticleSystem, ambientActive: number, perHit: number): void {
  ps.ambientActive = Math.max(0, Math.min(ps.ambient.cap, Math.floor(ambientActive)));
  ps.perHit = Math.max(0, Math.floor(perHit));
}

function burst(ps: ParticleSystem, pool: Pool, which: 0 | 1, x: number, y: number, count: number, power: number): void {
  if (pool.cap === 0) return;
  const n = Math.max(0, Math.floor(count));
  for (let k = 0; k < n; k++) {
    const i = ps.heads[which];
    ps.heads[which] = (i + 1) % pool.cap;
    const ang = rand(ps) * Math.PI * 2;
    const sp = power * (0.4 + rand(ps) * 0.6);
    pool.x[i] = x; pool.y[i] = y;
    pool.vx[i] = Math.cos(ang) * sp;
    pool.vy[i] = Math.sin(ang) * sp - power * 0.25;
    pool.life[i] = 0.25 + rand(ps) * 0.25;
  }
}

export function emitImpact(ps: ParticleSystem, x: number, y: number, count: number, power = 180): void {
  burst(ps, ps.impact, 0, x, y, count, power);
}
export function emitHurt(ps: ParticleSystem, x: number, y: number, count: number, power = 160): void {
  burst(ps, ps.hurt, 1, x, y, count, power);
}

function stepBurst(p: Pool, dt: number): void {
  for (let i = 0; i < p.cap; i++) {
    if (p.life[i] <= 0) continue;
    p.life[i] -= dt;
    if (p.life[i] <= 0) { p.life[i] = 0; continue; }
    p.x[i] += p.vx[i] * dt;
    p.vy[i] += 520 * dt;
    p.y[i] += p.vy[i] * dt;
  }
}

export function stepParticles(ps: ParticleSystem, dt: number, time: number): void {
  const a = ps.ambient;
  const n = Math.min(ps.ambientActive, a.cap);
  const rising = RISING[ps.style];
  const sway = ps.style === 'snow' || ps.style === 'embers';
  for (let i = 0; i < n; i++) {
    a.x[i] += (a.vx[i] + (sway ? Math.sin(time * 1.5 + i) * 8 : 0)) * dt;
    a.y[i] += a.vy[i] * dt;
    if (a.x[i] < 0) a.x[i] += WORLD_W; else if (a.x[i] > WORLD_W) a.x[i] -= WORLD_W;
    if (rising ? a.y[i] < -8 : a.y[i] > GROUND_Y) spawnAmbient(ps, i, false);
  }
  stepBurst(ps.impact, dt);
  stepBurst(ps.hurt, dt);
}

export function liveCount(p: Pool, limit = p.cap): number {
  let n = 0;
  for (let i = 0; i < Math.min(limit, p.cap); i++) if (p.life[i] > 0) n += 1;
  return n;
}

/** Live particles as drawable points (a shared off-screen point when there are none). */
export function pointsOf(p: Pool, limit = p.cap): Point[] {
  const out: Point[] = [];
  for (let i = 0; i < Math.min(limit, p.cap); i++) if (p.life[i] > 0) out.push({ x: p.x[i], y: p.y[i] });
  return out.length > 0 ? out : NO_POINTS;
}

import type { AttackDef } from './combat';
import type { Fighter } from './types';

/** lean/weapon in radians (positive = clockwise = forward/down); squash = vertical scale. */
export interface Pose { lean: number; weapon: number; squash: number }

const lerp = (a: number, b: number, k: number) => a + (b - a) * Math.min(1, Math.max(0, k));

/** Animation state machine output: a pose derived from state + time-in-state. */
export function poseFor(f: Fighter, def: AttackDef | null): Pose {
  switch (f.state) {
    case 'attack': {
      if (!def) return { lean: 0, weapon: 0.5, squash: 1 };
      if (f.t < def.windup) {
        const k = f.t / def.windup;
        return { lean: lerp(0, -0.15, k), weapon: lerp(0.5, -1.6, k), squash: 1 };
      }
      if (f.t < def.windup + def.active) {
        const k = (f.t - def.windup) / def.active;
        return { lean: lerp(-0.15, 0.25, k), weapon: lerp(-1.6, 1.2, k), squash: 1 };
      }
      const k = (f.t - def.windup - def.active) / def.recovery;
      return { lean: lerp(0.25, 0, k), weapon: lerp(1.2, 0.5, k), squash: 1 };
    }
    case 'telegraph': return { lean: -0.12, weapon: -1.9, squash: 1 };
    case 'block':     return { lean: -0.05, weapon: -0.4, squash: 1 };
    case 'hurt':      return { lean: -0.35, weapon: 1.0, squash: 1 };
    case 'dead':      return { lean: lerp(0, -1.45, f.t / 0.3), weapon: 1.2, squash: 1 };
    case 'dodge':     return { lean: 0.5, weapon: 0.8, squash: 0.82 };
    case 'special':   return { lean: 0, weapon: f.t * 20, squash: 1 };
    case 'recovery':  return { lean: 0.1, weapon: 0.5, squash: 1 };
    case 'walk':
    case 'approach':  return { lean: 0.06, weapon: 0.6, squash: 1 };
    default:          return { lean: 0, weapon: 0.5, squash: 1 };
  }
}

type Props = NativeStackScreenProps<RootStackParamList, 'Arena'>;
type OneShot = 'light' | 'heavy' | 'jump' | 'dodge' | 'special';
type Held = 'left' | 'right' | 'block';
interface Hud { combo: number; wave: number; step: number; boss: string; kinds: string }
interface Summary { stars: number; score: number; objectiveMet: boolean; reward: RewardBreakdown | null;  xp: number; levelUp: boolean; }

const sameHud = (a: Hud, b: Hud) =>
  a.combo === b.combo && a.wave === b.wave && a.step === b.step && a.boss === b.boss && a.kinds === b.kinds;

const readHud = (a: Arena): Hud => ({
  combo: a.combo.count, wave: a.wave, step: a.tutorial ? a.tutorial.index : -1,
  boss: bossLabel(a), kinds: a.enemies.map((e) => e.kind).join(','),
});

function makeArena(cfg: LevelConfig | null, q: Quality, perks?: ReturnType<typeof perksFor>): Arena {
  const p = PRESETS[q];
  const particles = {
    style: ambientStyleFor(cfg?.world ?? 1), ambient: p.ambient, impactCap: p.impactCap, perHit: p.impactPerHit,
  };
  if (!cfg) return createArena({ waves: [], tutorial: true, particles });
  return createArena({
    waves: cfg.waves, scaling: cfg.scaling, hazards: cfg.hazards, healPerWave: cfg.healPerWave,
    world: cfg.world, miniBoss: cfg.miniBoss, particles, perks,
  });
}

const ENEMY_SLOTS = Array.from({ length: MAX_ENEMIES }, (_, i) => i);
const PROJECTILE_SLOTS = Array.from({ length: MAX_PROJECTILES }, (_, i) => i);
const HAZARD_SLOTS = Array.from({ length: MAX_HAZARDS }, (_, i) => i);

export default function ArenaScreen({ navigation, route }: Props) {
  const params = route.params;
  const { save, recordResult } = useSave();
  const settings = save.settings;
  const vibrationRef = useRef(settings.vibration);
  vibrationRef.current = settings.vibration;
  const jumpHeld = useRef(false);
  const challenge = params.mode === 'level' && !!params.challenge;

  const [character] = useState(() => ({ visual: playerCharacterVisual(save) }));
  const [weapon] = useState(() => ({ visual: playerWeaponVisual(save) }));
  const [perks] = useState(() => combinePerks(perksFor(save.hero.ranks), gearPerks(save)));

  const cfg = useMemo<LevelConfig | null>(() => {
    if (params.mode !== 'level') return null;
    const base = getLevel(params.world, params.level);
    return params.challenge ? applyChallenge(base) : base;
  }, [params]);
  const worldIdx = params.mode === 'level' ? params.world : 1;
  const worldDef = WORLDS[worldIdx - 1];
  const palette = worldDef.palette;
  const bossInfo = cfg && (cfg.worldBoss || cfg.miniBoss) ? BOSSES[cfg.world] : null;

  const { width, height } = useWindowDimensions();
  const scale = Math.min(width / WORLD_W, height / WORLD_H);
  const offX = (width - WORLD_W * scale) / 2;
  const offY = (height - WORLD_H * scale) / 2;

  // Graphics: starts at the saved preset and can only be lowered during a session.
  const [quality, setQuality] = useState<Quality>(settings.graphics);
  const qualityRef = useRef<Quality>(settings.graphics);
  const preset = PRESETS[quality];
  const governor = useMemo(() => new PerfGovernor(1000 / settings.fps), [settings.fps]);
  const governorRef = useRef(governor);
  governorRef.current = governor;
  const minInterval = useRef(1000 / settings.fps - 2);
  const [toast, setToast] = useState('');

  const frameStats = useRef(new FrameStats());
  const [perfTxt, setPerfTxt] = useState('');

  const [initial] = useState(() => makeArena(cfg, settings.graphics, perks));
  const arena = useRef<Arena>(initial);
  const input = useRef<Input>(emptyInput());
  const paused = useRef(bossInfo !== null);
  const shown = useRef(false);
  const lastHud = useRef<Hud>(readHud(initial));

  const [intro, setIntro] = useState(bossInfo !== null);
  const [result, setResult] = useState<null | 'victory' | 'defeat'>(null);
  const [summary, setSummary] = useState<Summary | null>(null);
  const [isPaused, setIsPaused] = useState(false);
  const [hud, setHud] = useState<Hud>(lastHud.current);

  const onOver = useRef<(a: Arena) => void>(() => {});
  onOver.current = (a) => {
    if (!cfg) { setResult(a.over); return; }
    const run = evaluateRun(cfg, a);
    const levelBefore = levelFor(save.hero.xp);
    const out = recordResult(cfg, run, challenge, a.stats);
    const reward = run.won ? out : null;
    const xp = reward ? xpFromCoins(reward.total) : 0;
    setSummary({
      stars: run.stars, score: run.score, objectiveMet: run.objectiveMet, reward,
      xp, levelUp: levelFor(save.hero.xp + xp) > levelBefore,
    });
    setResult(a.over);
  };

  const onDowngrade = useRef<() => void>(() => {});
  onDowngrade.current = () => {
    const next = lowerQuality(qualityRef.current);
    if (next === qualityRef.current) return;
    qualityRef.current = next;
    setQuality(next);
    setParticleLimits(arena.current.particles, PRESETS[next].ambient, PRESETS[next].impactPerHit);
    setToast(`Graphics lowered to ${PRESETS[next].label} to keep the game smooth`);
  };

  useEffect(() => {
    if (!toast) return;
    const t = setTimeout(() => setToast(''), 3500);
    return () => clearTimeout(t);
  }, [toast]);

  useEffect(() => {
    playMusic(bossInfo ? 'boss' : (`world_${worldIdx}` as MusicTrack));
      return () => playMusic('menu');
  }, []); // eslint-disable-line react-hooks/exhaustive-deps

  const frame = useSharedValue<RenderFrame>(buildFrame(initial, character.visual.body));
  const shake = useSharedValue(0);
  const ambientPts = useSharedValue<Point[]>(NO_POINTS);
  const impactPts = useSharedValue<Point[]>(NO_POINTS);
  const hurtPts = useSharedValue<Point[]>(NO_POINTS);

  const transform = useDerivedValue(() => [
    { translateX: offX + shake.value * scale },
    { translateY: offY },
    { scale },
  ]);
  const pHpW = useDerivedValue(() => frame.value.playerHp * 200);
  const eHpW = useDerivedValue(() => frame.value.enemyHp * 200);
  const enW = useDerivedValue(() => frame.value.energy * 120);
  const enColor = useDerivedValue(() => (frame.value.energy >= 0.5 ? '#35d0ff' : '#2a6f8a'));
  const bossW = useDerivedValue(() => Math.max(0, frame.value.bossHp) * 280);
  const bossOn = useDerivedValue(() => (frame.value.bossHp >= 0 ? 1 : 0));

  const sync = useCallback(() => {
    const a = arena.current;
    frame.value = buildFrame(a, character.visual.body);
    shake.value = a.shake > 0 ? Math.sin(a.time * 90) * a.shake : 0;

    const ps = a.particles;
    if (ps.ambientActive > 0) ambientPts.value = pointsOf(ps.ambient, ps.ambientActive);
    if (ps.impact.cap > 0) {
      impactPts.value = pointsOf(ps.impact);
      hurtPts.value = pointsOf(ps.hurt);
    }

    if (a.events.length > 0) {
      for (const ev of a.events) { playSfx(ev); if (vibrationRef.current) buzz(ev); }
      a.events.length = 0;
    }

    const next = readHud(a); // React state only changes when something visible changes (rare)
    if (!sameHud(lastHud.current, next)) { lastHud.current = next; setHud(next); }
    if (a.over && !shown.current) { shown.current = true; onOver.current(a); }
  }, []); // eslint-disable-line react-hooks/exhaustive-deps

  useEffect(() => {
    let raf = 0;
    let last = performance.now();
    let lastSync = 0;
    let lastPerf = last;
    let acc = 0;
    const loop = (now: number) => {
      const ms = now - last;
      const dt = Math.min(ms / 1000, 0.1);
      last = now;
      if (SHOW_PERF && !paused.current) {
        frameStats.current.add(ms);
        if (now - lastPerf >= 1000) {
          lastPerf = now;
          const s = frameStats.current.summary();
          setPerfTxt(`${s.fps.toFixed(0)} fps · avg ${s.avgMs.toFixed(1)}ms · p95 ${s.p95Ms.toFixed(1)}ms · worst ${s.worstMs.toFixed(0)}ms · ${qualityRef.current}`);
        }
      }
      if (!paused.current && !arena.current.over) {
        acc += dt;
        while (acc >= FIXED_DT) {
          stepArena(arena.current, input.current, FIXED_DT);
          const i = input.current; // one-shot buttons are consumed after the first sub-step
          i.light = i.heavy = i.jump = i.dodge = i.special = false;
          acc -= FIXED_DT;
        }
        if (governorRef.current.sample(ms)) onDowngrade.current();
      }
      if (now - lastSync >= minInterval.current) { lastSync = now; sync(); } // 30 FPS option skips frames
      raf = requestAnimationFrame(loop);
    };
    raf = requestAnimationFrame(loop);
    return () => cancelAnimationFrame(raf);
  }, [sync]);

  const startFight = () => { paused.current = false; setIntro(false); };

  const retry = () => {
    arena.current = makeArena(cfg, qualityRef.current, perks);
    input.current = emptyInput();
    shown.current = false;
    paused.current = false;
    lastHud.current = readHud(arena.current);
    setHud(lastHud.current);
    setSummary(null);
    setIsPaused(false);
    setResult(null);
  };

  const hold = (k: Held) => ({
    onPressIn: () => { input.current[k] = true; },
    onPressOut: () => { input.current[k] = false; },
  });
  const tap = (k: OneShot) => ({ onPressIn: () => { input.current[k] = true; } });

  const onStick = (dx: number, dy: number) => {
    const r = joystickToInput(dx, dy, jumpHeld.current);
    jumpHeld.current = r.jumpHeld;
    const i = input.current;
    i.left = r.left; i.right = r.right; i.block = r.block;
    if (r.jump) i.jump = true;
  };

  const togglePause = () => {
    if (intro) return;
    paused.current = !paused.current;
    input.current = emptyInput();
    setIsPaused(paused.current);
  };

  const nextTarget = useMemo(() => {
    if (params.mode !== 'level' || challenge) return null;
    if (params.level < 100) return { world: params.world, level: params.level + 1 };
    return params.world < 5 ? { world: params.world + 1, level: 1 } : null;
  }, [params, challenge]);
  const canNext = result === 'victory' && nextTarget !== null && isLevelUnlocked(save, nextTarget.world, nextTarget.level);

  const kinds = hud.kinds.split(',');
  const ambient = AMBIENT_LOOK[ambientStyleFor(worldIdx)];
  const cs = settings.controlSize;
  const prompt = params.mode === 'tutorial' ? TUTORIAL_STEPS[hud.step]?.prompt : undefined;
  const title = result === 'victory'
    ? (params.mode === 'tutorial' ? 'TUTORIAL COMPLETE' : 'VICTORY')
    : result === 'defeat' ? 'DEFEAT' : 'PAUSED';
  const r = summary?.reward;
  const introName = bossInfo
    ? (cfg?.miniBoss ? `CHAMPION OF ${bossInfo.name.toUpperCase()}` : bossInfo.name.toUpperCase())
    : '';

  return (
    <View style={styles.root}>
      <Canvas style={StyleSheet.absoluteFill}>
        <Group transform={transform}>
          {/* PLACEHOLDER ART: flat shapes and polygon silhouettes tinted with the world palette */}
          <Rect x={0} y={0} width={WORLD_W} height={WORLD_H} color={palette.sky} />
          <Circle cx={650} cy={70} r={38} color={palette.moon} opacity={0.85} />
          <Background world={worldDef} layers={preset.parallax} fog={preset.fog} frame={frame} />
          <Rect x={0} y={GROUND_Y} width={WORLD_W} height={WORLD_H - GROUND_Y} color={palette.ground} />

          {preset.ambient > 0 && (
            <Points points={ambientPts} mode="points" color={ambient.color} style="stroke"
                    strokeWidth={ambient.size} strokeCap="square" opacity={ambient.opacity} />
          )}

          {HAZARD_SLOTS.map((i) => <HazardView key={i} frame={frame} index={i} />)}
          {ENEMY_SLOTS.map((i) => {
            const kind = (kinds[i] || 'melee') as EnemyKind;
            return (
              <FighterView key={`${i}-${kind}`} frame={frame} who={i} glow={preset.glow}
                           look={enemyLookFor(kind, worldIdx)} weapon={enemyWeaponFor(kind)} />
            );
          })}
          <FighterView frame={frame} who="player" look={character.visual} weapon={weapon.visual} glow={preset.glow} />
          {preset.hitFx && <HitFxView frame={frame} style={weapon.visual.hitFx} color={weapon.visual.hitColor} />}
          {PROJECTILE_SLOTS.map((i) => <ProjectileView key={i} frame={frame} index={i} />)}

          <Points points={impactPts} mode="points" color={weapon.visual.hitColor} style="stroke"
                  strokeWidth={3} strokeCap="square" />
          <Points points={hurtPts} mode="points" color="#ff4d5e" style="stroke"
                  strokeWidth={3} strokeCap="square" />

          <Circle cx={24} cy={26} r={17} color={character.visual.body} />
          <Circle cx={24} cy={26} r={17} color={character.visual.accent} style="stroke" strokeWidth={2} />
          <Rect x={48} y={14} width={200} height={10} color="#222" />
          <Rect x={48} y={14} width={pHpW} height={10} color="#4cd964" />
          <Rect x={48} y={28} width={120} height={6} color="#222" />
          <Rect x={48} y={28} width={enW} height={6} color={enColor} />
          <Circle cx={776} cy={26} r={17} color="#b3263e" />
          <Circle cx={776} cy={26} r={17} color="#ff8a9a" style="stroke" strokeWidth={2} />
          <Rect x={552} y={14} width={200} height={10} color="#222" />
          <Rect x={552} y={14} width={eHpW} height={10} color="#e0334c" />
          <Rect x={260} y={42} width={280} height={8} color="#222" opacity={bossOn} />
          <Rect x={260} y={42} width={bossW} height={8} color="#ff8c1a" opacity={bossOn} />
        </Group>
      </Canvas>

      {cfg && (
        <>
          <Text style={styles.wave}>{challenge ? 'CHALLENGE · ' : ''}{cfg.id.toUpperCase()}  WAVE {hud.wave}/{cfg.waves.length}</Text>
          <Text style={styles.objective}>★ {cfg.objective.label}</Text>
        </>
      )}
      {hud.boss !== '' && <Text style={styles.bossName}>{hud.boss}</Text>}
      {hud.combo >= 2 && <Text style={styles.combo}>{hud.combo} HIT COMBO</Text>}
      {prompt && (
        <View style={styles.prompt}>
          <Text style={styles.promptTxt}>{hud.step + 1}/{TUTORIAL_STEPS.length}  {prompt}</Text>
        </View>
      )}
      {!!toast && <View style={styles.toast}><Text style={styles.toastTxt}>{toast}</Text></View>}


      {SHOW_PERF && !!perfTxt && <Text style={styles.perf}>{perfTxt}</Text>}

            {settings.controls === 'joystick' ? (
        <View style={[styles.cluster, { left: 24 }]}>
          <Joystick size={Math.round(124 * cs)} onChange={onStick} />
        </View>
      ) : (
        <View style={[styles.cluster, { left: 24, flexDirection: 'row' }]}>
          <Btn label="◀" size={64 * cs} {...hold('left')} />
          <Btn label="▶" size={64 * cs} {...hold('right')} />
        </View>
      )}
      {cfg && (
        <View style={[styles.badge, { left: width / 2 + 38 }]}>
          <Text style={styles.badgeTxt}>{cfg.level}</Text>
        </View>
      )}
      <View style={[styles.cluster, { right: 24 }]}>
        <View style={styles.row}>
          <Btn label="BLOCK" size={54 * cs} {...hold('block')} />
          <Btn label="DODGE" size={54 * cs} {...tap('dodge')} />
          <Btn label="JUMP" size={54 * cs} {...tap('jump')} />
        </View>
        <View style={styles.row}>
          <Btn label="SPECIAL" size={64 * cs} {...tap('special')} />
          <Btn label="HEAVY" size={76 * cs} {...tap('heavy')} />
          <Btn label="ATK" size={76 * cs} {...tap('light')} />
        </View>
      </View>

      <Pressable style={styles.pause} onPress={togglePause}>
        <Text style={styles.pauseTxt}>{isPaused ? '▶' : 'II'}</Text>
      </Pressable>

      {intro && bossInfo && (
        <Pressable style={styles.intro} onPress={startFight}>
          <Text style={styles.introTag}>{cfg?.worldBoss ? 'WORLD BOSS' : 'MINI-BOSS'}</Text>
          <Text style={[styles.introName, { color: palette.accent }]}>{introName}</Text>
          <Text style={styles.introTitle}>{bossInfo.title}</Text>
          <Text style={styles.introBlurb}>{bossInfo.blurb}</Text>
          <Text style={styles.introBlurb}>
            {cfg?.miniBoss ? 2 : bossInfo.phases.length} phases · wait for the yellow flash, then dodge or block
          </Text>
          <Text style={styles.introTap}>TAP TO FIGHT</Text>
        </Pressable>
      )}

            {isPaused && !result && (
        <PauseMenu
          onResume={togglePause}
          onRestart={retry}
          onSettings={() => navigation.navigate('Settings')}
          onMenu={() => navigation.navigate('Home')}
        />
      )}
      {result === 'victory' && (
        <VictoryPanel
          title={title}
          stars={summary?.stars}
          score={summary?.score}
          objectiveMet={summary?.objectiveMet}
          reward={summary?.reward ?? null}
          onNext={canNext && nextTarget
            ? () => navigation.replace('Arena', { mode: 'level', world: nextTarget.world, level: nextTarget.level })
            : undefined}
          onRetry={retry}
          onExit={() => navigation.goBack()}
          exitLabel={cfg ? 'LEVELS' : 'HOME'}
          xp={summary?.xp}
          levelUp={summary?.levelUp}
        />
      )}
      {result === 'defeat' && (
        <DefeatPanel onRetry={retry} onExit={() => navigation.goBack()} exitLabel={cfg ? 'LEVELS' : 'HOME'} />
      )}
    </View>
  );
}

function Btn(props: { label: string; size: number; onPressIn: () => void; onPressOut?: () => void }) {
  const s = Math.round(props.size);
  return (
    <Pressable
      onPressIn={props.onPressIn}
      onPressOut={props.onPressOut}
      style={({ pressed }) => [styles.btn, { width: s, height: s, borderRadius: s / 2 }, pressed && styles.btnDown]}
    >
      <Text style={styles.btnTxt}>{props.label}</Text>
    </Pressable>
  );
}

import React from 'react';
import { Circle, Group, Rect } from '@shopify/react-native-skia';
import type { SharedValue } from 'react-native-reanimated';
import { FIGHTER_H, FIGHTER_W } from '../engine/constants';
import type { CharacterFeature, CharacterVisual, WeaponVisual } from '../cosmetics/types';

type ColorProp = string | SharedValue<string>;

/**
 * PLACEHOLDER ART: layered shapes in local coordinates (origin = feet, +x = facing).
 * Purely visual. Extras like capes can poke slightly outside the hitbox, which never changes.
 */
export function CharacterArt({ look, color }: { look: CharacterVisual; color: ColorProp }) {
  const W = FIGHTER_W * look.widthScale;
  const bodyH = FIGHTER_H * 0.78 * look.heightScale;
  const headR = FIGHTER_H * 0.12 * look.headScale;
  const headY = -bodyH - headR * 0.83;
  const has = (f: CharacterFeature) => look.features.includes(f);

  return (
    <>
      {has('cape') && <Rect x={-W / 2 - 12} y={-bodyH * 0.95} width={16} height={bodyH * 0.9} color={look.accent} opacity={0.9} />}
      {has('tail') && <Rect x={-W / 2 - 18} y={-bodyH * 0.3} width={20} height={5} color={look.accent} />}
      {has('scarf') && <Rect x={-W / 2 - 16} y={-bodyH - 2} width={24} height={6} color={look.trim} />}
      {has('spikes') && [0, 1, 2].map((i) => (
        <Rect key={i} x={-W / 2 + 3 + i * 10} y={-bodyH - 9} width={4} height={10} color={look.accent} />

      {has('pauldrons') && (

          <Rect x={W / 2 - 7} y={-bodyH - 1} width={13} height={11} color={look.accent} />

      )}

      {has('hood') && <Circle cx={0} cy={headY} r={headR * 1.3} color={look.accent} />}
      {has('halo') && <Circle cx={0} cy={headY - headR * 1.9} r={headR * 0.9} color={look.accent} style="stroke" strokeWidth={3} />}
      <Circle cx={0} cy={headY} r={headR} color={color} />
      {has('hat') && (

          <Rect x={-headR * 1.9} y={headY - headR - 1} width={headR * 3.8} height={4} color={look.accent} />

  );
}

/** Weapon drawn along +x from the hand (origin). `length` is drawing only; it never changes attack reach. */
export function WeaponArt({ weapon, glow = true }: { weapon: WeaponVisual; glow?: boolean }) {
  const L = weapon.length;
  const { blade, handle } = weapon;
  const weaponGlow = weapon.glow;
  const halo = glow ? <Rect x={-4} y={-7} width={L + 8} height={14} color={weaponGlow} opacity={0.28} /> : null;

  switch (weapon.shape) {
    case 'katana':
      return (<>{halo}
        <Rect x={-10} y={-2.5} width={12} height={5} color={handle} />
        <Rect x={2} y={-6} width={3} height={12} color={weaponGlow} />
        <Rect x={5} y={-2} width={L - 5} height={4} color={blade} /></>);
    case 'greatsword':
      return (<>{halo}
        <Rect x={-12} y={-3} width={14} height={6} color={handle} />
        <Rect x={2} y={-8} width={4} height={16} color={weaponGlow} />
        <Rect x={6} y={-5} width={L - 6} height={10} color={blade} /></>);
    case 'dual':
      return (<>{halo}
        <Rect x={-6} y={-3} width={10} height={6} color={handle} />
        <Rect x={4} y={-7} width={L * 0.85} height={3} color={blade} />
        <Rect x={4} y={4} width={L * 0.85} height={3} color={blade} /></>);
    case 'daggers':
      return (<>
        <Rect x={-6} y={-3} width={10} height={6} color={handle} />
        <Rect x={4} y={-8} width={L * 0.6} height={3} color={blade} />
        <Rect x={4} y={5} width={L * 0.6} height={3} color={blade} />
        <Rect x={4} y={-8} width={L * 0.6} height={3} color={weaponGlow} opacity={0.35} />
        <Rect x={4} y={5} width={L * 0.6} height={3} color={weaponGlow} opacity={0.35} /></>);
    case 'spear':
      return (<>{halo}
        <Rect x={-14} y={-1.5} width={L} height={3} color={handle} />
        <Group transform={[{ translateX: L }, { rotate: Math.PI / 4 }]}>
          <Rect x={-5} y={-5} width={10} height={10} color={blade} />
        </Group></>);
    case 'scythe':
      return (<>
        <Rect x={-12} y={-2} width={L} height={4} color={handle} />
        <Rect x={L - 18} y={-26} width={7} height={28} color={blade} />
        <Rect x={L - 18} y={-26} width={7} height={28} color={weaponGlow} opacity={0.3} />
        <Group transform={[{ translateX: L - 16 }, { translateY: -26 }, { rotate: 0.6 }]}>
          <Rect x={0} y={-3} width={14} height={6} color={blade} />
        </Group></>);
    case 'chain':
      return (<>
        <Rect x={-8} y={-2} width={10} height={4} color={handle} />
        {[0, 1, 2, 3].map((i) => <Circle key={i} cx={6 + i * L * 0.17} cy={0} r={3} color={handle} />)}
        <Rect x={L * 0.7} y={-5} width={L * 0.3} height={10} color={weaponGlow} opacity={0.35} />
        <Rect x={L * 0.7} y={-4} width={L * 0.3} height={8} color={blade} /></>);
    case 'axe':
      return (<>
        <Rect x={-8} y={-2} width={L + 8} height={4} color={handle} />
        <Rect x={L - 16} y={-15} width={14} height={30} color={blade} />
        <Rect x={L - 2} y={-15} width={3} height={30} color={weaponGlow} /></>);
    case 'staff':
      return (<>
        <Rect x={-10} y={-2} width={L + 10} height={4} color={handle} />
        <Circle cx={L + 4} cy={0} r={11} color={weaponGlow} opacity={0.35} />
        <Circle cx={L + 4} cy={0} r={6} color={blade} /></>);
    default:
      return null;
  }
}

const SPARKS = 6;

type Fx = { frame: SharedValue<RenderFrame>; color: string };

function Spark({ frame, color, i, style }: Fx & { i: number; style: HitFxStyle }) {
  const angle = (i / SPARKS) * Math.PI * 2 + (style === 'frost' ? Math.PI / 6 : 0);
  const t = useDerivedValue(() => {
    const fx = frame.value.fx;
    const ease = 1 - (1 - fx.k) * (1 - fx.k);
    const dist = (style === 'frost' ? 26 : 34) * ease;
    const rise = style === 'ember' ? -fx.k * 22 : 0; // embers drift upward
    return [
      { translateX: fx.x + Math.cos(angle) * dist },
      { translateY: fx.y + Math.sin(angle) * dist + rise },
      { rotate: angle },
    ];
  });
  const op = useDerivedValue(() => frame.value.fx.on * (1 - frame.value.fx.k));
  const w = style === 'frost' ? 6 : style === 'ember' ? 4 : 10;
  const h = style === 'frost' ? 6 : style === 'ember' ? 4 : 3;
  return (
    <Group transform={t} opacity={op}>
      <Rect x={-w / 2} y={-h / 2} width={w} height={h} color={color} />
    </Group>
  );
}

function Ring({ frame, color }: Fx) {
  const cx = useDerivedValue(() => frame.value.fx.x);
  const cy = useDerivedValue(() => frame.value.fx.y);
  const r = useDerivedValue(() => 6 + frame.value.fx.k * 34);
  const op = useDerivedValue(() => frame.value.fx.on * (1 - frame.value.fx.k));
  return <Circle cx={cx} cy={cy} r={r} color={color} style="stroke" strokeWidth={3} opacity={op} />;
}

    { translateX: frame.value.fx.x },
    { translateY: frame.value.fx.y },
    { rotate: angle },
    { scaleX: 0.3 + frame.value.fx.k },
  ]);
  const t1 = make(0.7);
  const t2 = make(-0.7);
  const op = useDerivedValue(() => frame.value.fx.on * (1 - frame.value.fx.k));
  return (
    <>
      <Group transform={t1} opacity={op}><Rect x={-30} y={-1.5} width={60} height={3} color={color} /></Group>
      <Group transform={t2} opacity={op}><Rect x={-30} y={-1.5} width={60} height={3} color={color} /></Group>
    </>
  );
}

/** Which layers to draw for a given count: 1 = the middle one, 2 = far + near, 3 = all. */
export function pickLayers(defs: LayerDef[], count: number): LayerDef[] {
  if (count <= 1) return [defs[1] ?? defs[0]];
  if (count === 2) return [defs[0], defs[defs.length - 1]];
  return defs;
}

export function mixHex(a: string, b: string, t: number): string {
  const pa = parseInt(a.slice(1), 16);
  const pb = parseInt(b.slice(1), 16);
  const ch = (shift: number) =>
    Math.round(((pa >> shift) & 255) * (1 - t) + ((pb >> shift) & 255) * t);
  return `#${((1 << 24) | (ch(16) << 16) | (ch(8) << 8) | ch(0)).toString(16).slice(1)}`;
}

function lcg(seed: number): () => number {
  let s = seed >>> 0;
  return () => {
    s = (Math.imul(s, 1664525) + 1013904223) >>> 0;
    return s / 4294967296;
  };
}

const f = (n: number) => Math.round(n * 10) / 10;

/** One SVG path string for a whole layer. Deterministic for a given seed. */
export function layerPath(def: LayerDef, seed: number, x0 = -420, x1 = 1220): string {
  const rnd = lcg(seed * 7919 + 13);
  const B = def.baseY;
  const parts: string[] = [];
  let x = x0;
  while (x < x1) {
    const h = def.minH + rnd() * (def.maxH - def.minH);
    switch (def.kind) {
      case 'peaks': {
        const w = 90 + rnd() * 70;
        parts.push(`M${f(x)} ${B}L${f(x + w / 2)} ${f(B - h)}L${f(x + w)} ${B}Z`);
        x += w * 0.6; break;
      }
      case 'spires': {
        const w = 18 + rnd() * 16;
        parts.push(`M${f(x)} ${B}L${f(x + w / 2)} ${f(B - h)}L${f(x + w)} ${B}Z`);
        x += w + 30 + rnd() * 40; break;
      }
      case 'trees': {
        const w = 34 + rnd() * 22;
        const cx = x + w / 2;
        parts.push(`M${f(x)} ${f(B - h * 0.25)}L${f(cx)} ${f(B - h * 0.7)}L${f(x + w)} ${f(B - h * 0.25)}Z`);
        parts.push(`M${f(x + w * 0.15)} ${f(B - h * 0.55)}L${f(cx)} ${f(B - h)}L${f(x + w * 0.85)} ${f(B - h * 0.55)}Z`);
        parts.push(`M${f(cx - 3)} ${B}L${f(cx - 3)} ${f(B - h * 0.3)}L${f(cx + 3)} ${f(B - h * 0.3)}L${f(cx + 3)} ${B}Z`);
        x += w + 14 + rnd() * 26; break;
      }
      case 'crystals': {
        const w = 20 + rnd() * 20;
        const cx = x + w / 2;
        parts.push(`M${f(cx)} ${f(B - h)}L${f(x + w)} ${f(B - h * 0.45)}L${f(cx)} ${B}L${f(x)} ${f(B - h * 0.45)}Z`);
        x += w + 20 + rnd() * 40; break;
      }
      case 'towers': {
        const w = 40 + rnd() * 30;
        const c = w / 5;
        parts.push(`M${f(x)} ${B}L${f(x)} ${f(B - h)}L${f(x + w)} ${f(B - h)}L${f(x + w)} ${B}Z`);
        for (let k = 0; k < 3; k++) {
          const cx = x + c * (k * 2);
          parts.push(`M${f(cx)} ${f(B - h)}L${f(cx)} ${f(B - h - 10)}L${f(cx + c)} ${f(B - h - 10)}L${f(cx + c)} ${f(B - h)}Z`);
        }
        x += w + 20 + rnd() * 50; break;
      }
      case 'shards': {
        const w = 24 + rnd() * 30;
        const y = B - h - rnd() * 60;
        parts.push(`M${f(x)} ${f(y)}L${f(x + w)} ${f(y)}L${f(x + w / 2)} ${f(y + h)}Z`);
        x += w + 60 + rnd() * 80; break;
      }
    }
  }
  return parts.join('');
}

========== src/cosmetics/types.ts ==========
export type Rarity = 'common' | 'rare' | 'epic' | 'legendary';
export type ItemKind = 'character' | 'weapon';

export const RARITY_ORDER: Rarity[] = ['common', 'rare', 'epic', 'legendary'];
export const RARITY_COLOR: Record<Rarity, string> = {
  common: '#9aa3b5', rare: '#4da3ff', epic: '#b66bff', legendary: '#ffb524',
};

/*
 * COSMETIC-ONLY RULE: skins describe how something LOOKS. They must never contain
 * damage, reach, speed, health or any other gameplay number. A test enforces the key list.
 */

export type CharacterFeature =
  | 'hood' | 'horns' | 'cape' | 'hat' | 'mask' | 'spikes' | 'halo' | 'scarf' | 'pauldrons' | 'tail';

export interface CharacterVisual {
  body: string; accent: string; trim: string; eye: string;
  widthScale: number; heightScale: number; headScale: number; // 0.9..1.1, drawing only
  features: CharacterFeature[];
}

export interface CharacterSkin {
  id: string; name: string; rarity: Rarity; free?: boolean; blurb: string; visual: CharacterVisual;
}

export type WeaponShape =
  | 'katana' | 'dual' | 'daggers' | 'spear' | 'scythe' | 'chain' | 'axe' | 'staff' | 'greatsword';
export type HitFxStyle = 'spark' | 'ring' | 'slash' | 'ember' | 'frost';

export interface WeaponVisual {
  shape: WeaponShape;
  length: number;     // drawing length in px (28..56). Does NOT change attack reach.
  blade: string; handle: string; glow: string;
  trail: string;      // color of the slash trail
  hitFx: HitFxStyle; hitColor: string;
}

export interface WeaponSkin {
  id: string; name: string; rarity: Rarity; free?: boolean; blurb: string; visual: WeaponVisual;
}
